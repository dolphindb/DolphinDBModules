#include "setup/settings.txt"
use MockData

@testing:case="test_MockData_function_XBondDepth_syntax"
re = syntax(XBondDepth)
ex = "XBondDepth(tradeDate,securityNumber)"
assert 1, eqObj(re, ex)

@testing:case="test_MockData_function_XBondDepth_normal"
t = XBondDepth(tradeDate=2020.06.01, securityNumber=10)
assert 1, form(t)==6
assert 2, t.rows()==10000
assert 3, t.cols()==33
assert 4, eqObj(t.schema().colDefs[0].name, "bondCodeVal")
assert 5, eqObj(t.schema().colDefs[0].typeInt, 17)

@testing:case="test_MockData_function_XBondDepth_default_securityNumber"
t = XBondDepth(tradeDate=2020.06.01)
assert 1, form(t)==6
assert 2, t.cols()==33

@testing:case="test_MockData_function_XBondDepth_securityNumber_1"
t = XBondDepth(tradeDate=2020.06.01, securityNumber=1)
assert 1, t.rows()==1000
assert 2, t.cols()==33

@testing:case="test_MockData_function_XBondDepth_non_trading_day"
t = XBondDepth(tradeDate=2020.06.07, securityNumber=10)
assert 1, t.rows()==0
assert 2, t.cols()==33

@testing:case="test_MockData_function_XBondDepth_invalid_tradeDate_int", exception=1
XBondDepth(tradeDate=1, securityNumber=10)

@testing:case="test_MockData_function_XBondDepth_invalid_tradeDate_string", exception=1
XBondDepth(tradeDate="2020.06.01", securityNumber=10)

@testing:case="test_MockData_function_XBondDepth_invalid_securityNumber_string", exception=1
XBondDepth(tradeDate=2020.06.01, securityNumber="10")

@testing:case="test_MockData_function_XBondDepth_invalid_securityNumber_double", exception=1
XBondDepth(tradeDate=2020.06.01, securityNumber=1.1)
