#include "setup/settings.txt"
use MockData

@testing:case="test_MockData_function_halfHourFactor_syntax"
re = syntax(halfHourFactor)
ex = "halfHourFactor(startDate,endDate,securityNumber,factorNumber)"
assert 1, eqObj(re, ex)

@testing:case="test_MockData_function_halfHourFactor_normal"
t = halfHourFactor(startDate=2021.12.30, endDate=2021.12.31, securityNumber=10, factorNumber=5)
assert 1, form(t)==6
assert 2, t.cols()==4
assert 3, eqObj(t.schema().colDefs[0].name, "tradetime")
assert 4, eqObj(t.schema().colDefs[0].typeInt, 12)
assert 5, eqObj(t.schema().colDefs[1].name, "securityid")
assert 6, eqObj(t.schema().colDefs[1].typeInt, 17)
assert 7, eqObj(t.schema().colDefs[2].name, "value")
assert 8, eqObj(t.schema().colDefs[2].typeInt, 16)
assert 9, eqObj(t.schema().colDefs[3].name, "factorname")
assert 10, eqObj(t.schema().colDefs[3].typeInt, 17)

@testing:case="test_MockData_function_halfHourFactor_one_day_one_stock_one_factor"
t = halfHourFactor(startDate=2021.12.31, endDate=2021.12.31, securityNumber=1, factorNumber=1)
assert 1, t.rows()==8
assert 2, t.cols()==4

@testing:case="test_MockData_function_halfHourFactor_non_trading_day_range"
t = halfHourFactor(startDate=2021.01.01, endDate=2021.01.03, securityNumber=10, factorNumber=5)
assert 1, t.rows()==0
assert 2, t.cols()==4

@testing:case="test_MockData_function_halfHourFactor_endDate_before_startDate", exception=1
halfHourFactor(startDate=2021.12.31, endDate=2021.12.30, securityNumber=10, factorNumber=5)

@testing:case="test_MockData_function_halfHourFactor_invalid_startDate_int", exception=1
halfHourFactor(startDate=1, endDate=2021.12.31, securityNumber=10, factorNumber=5)

@testing:case="test_MockData_function_halfHourFactor_invalid_securityNumber_string", exception=1
halfHourFactor(startDate=2021.12.30, endDate=2021.12.31, securityNumber="10", factorNumber=5)

@testing:case="test_MockData_function_halfHourFactor_invalid_factorNumber_string", exception=1
halfHourFactor(startDate=2021.12.30, endDate=2021.12.31, securityNumber=10, factorNumber="5")
