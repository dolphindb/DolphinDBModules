#include "setup/settings.txt"
use MockData

@testing:case="test_MockData_function_stockEntrust_syntax"
re = syntax(stockEntrust)
ex = "stockEntrust(tradeDate,securityNumber)"
assert 1, eqObj(re, ex)

@testing:case="test_MockData_function_stockEntrust_normal"
t = stockEntrust(tradeDate=2020.01.06, securityNumber=10)
assert 1, form(t)==6
assert 2, t.rows()==288020
assert 3, t.cols()==16
assert 4, eqObj(t.schema().colDefs[0].name, "ChannelNo")
assert 5, eqObj(t.schema().colDefs[3].name, "SecurityID")
assert 6, eqObj(t.schema().colDefs[3].typeInt, 17)

@testing:case="test_MockData_function_stockEntrust_securityNumber_1"
t = stockEntrust(tradeDate=2020.01.06, securityNumber=1)
assert 1, t.rows()==28802
assert 2, t.cols()==16

@testing:case="test_MockData_function_stockEntrust_non_trading_day"
t = stockEntrust(tradeDate=2020.01.05, securityNumber=10)
assert 1, t.rows()==0
assert 2, t.cols()==16

@testing:case="test_MockData_function_stockEntrust_invalid_tradeDate_int", exception=1
stockEntrust(tradeDate=1, securityNumber=10)

@testing:case="test_MockData_function_stockEntrust_invalid_tradeDate_string", exception=1
stockEntrust(tradeDate="2020.01.06", securityNumber=10)

@testing:case="test_MockData_function_stockEntrust_invalid_securityNumber_string", exception=1
stockEntrust(tradeDate=2020.01.06, securityNumber="10")

@testing:case="test_MockData_function_stockEntrust_invalid_securityNumber_double", exception=1
stockEntrust(tradeDate=2020.01.06, securityNumber=1.1)

@testing:case="test_MockData_function_stockEntrust_invalid_securityNumber_NULL", exception=1
stockEntrust(tradeDate=2020.01.06, securityNumber=NULL)
