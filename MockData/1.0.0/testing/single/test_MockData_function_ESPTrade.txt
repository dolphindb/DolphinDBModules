#include "setup/settings.txt"
use MockData

@testing:case="test_MockData_function_ESPTrade_syntax"
re = syntax(ESPTrade)
ex = "ESPTrade(tradeDate,securityNumber)"
assert 1, eqObj(re, ex)

@testing:case="test_MockData_function_ESPTrade_normal"
t = ESPTrade(tradeDate=2020.06.01, securityNumber=10)
assert 1, form(t)==6
assert 2, t.rows()==100
assert 3, t.cols()==20
assert 4, eqObj(t.schema().colDefs[0].name, "createDate")
assert 5, eqObj(t.schema().colDefs[0].typeInt, 6)

@testing:case="test_MockData_function_ESPTrade_default_securityNumber"
t = ESPTrade(tradeDate=2020.06.01)
assert 1, form(t)==6
assert 2, t.cols()==20

@testing:case="test_MockData_function_ESPTrade_securityNumber_1"
t = ESPTrade(tradeDate=2020.06.01, securityNumber=1)
assert 1, t.rows()==10
assert 2, t.cols()==20

@testing:case="test_MockData_function_ESPTrade_non_trading_day"
t = ESPTrade(tradeDate=2020.06.07, securityNumber=10)
assert 1, t.rows()==0
assert 2, t.cols()==20

@testing:case="test_MockData_function_ESPTrade_invalid_tradeDate_int", exception=1
ESPTrade(tradeDate=1, securityNumber=10)

@testing:case="test_MockData_function_ESPTrade_invalid_tradeDate_string", exception=1
ESPTrade(tradeDate="2020.06.01", securityNumber=10)

@testing:case="test_MockData_function_ESPTrade_invalid_securityNumber_string", exception=1
ESPTrade(tradeDate=2020.06.01, securityNumber="10")

@testing:case="test_MockData_function_ESPTrade_invalid_securityNumber_double", exception=1
ESPTrade(tradeDate=2020.06.01, securityNumber=1.1)
