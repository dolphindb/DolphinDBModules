#include "setup/settings.txt"
use MockData

@testing:case="test_MockData_function_futuresFactor_syntax"
re = syntax(futuresFactor)
ex = "futuresFactor(startDate,endDate,securityNumber,factorNumber)"
assert 1, eqObj(re, ex)

@testing:case="test_MockData_function_futuresFactor_normal"
t = futuresFactor(startDate=2021.12.31, endDate=2021.12.31, securityNumber=1, factorNumber=1)
assert 1, form(t)==6
assert 2, t.cols()==4
assert 3, eqObj(t.schema().colDefs[0].name, "tradetime")
assert 4, eqObj(t.schema().colDefs[0].typeInt, 12)
assert 5, eqObj(t.schema().colDefs[1].name, "securityid")
assert 6, eqObj(t.schema().colDefs[1].typeInt, 17)
assert 7, eqObj(t.schema().colDefs[2].name, "value")
assert 8, eqObj(t.schema().colDefs[2].typeInt, 16)
assert 9, eqObj(t.schema().colDefs[3].name, "factorname")
assert 10, eqObj(t.schema().colDefs[3].typeInt, 17)

@testing:case="test_MockData_function_futuresFactor_one_day_one_stock_one_factor"
t = futuresFactor(startDate=2021.12.31, endDate=2021.12.31, securityNumber=1, factorNumber=1)
assert 1, t.rows()==66600
assert 2, t.cols()==4

@testing:case="test_MockData_function_futuresFactor_non_trading_day_range"
t = futuresFactor(startDate=2021.01.01, endDate=2021.01.03, securityNumber=1, factorNumber=1)
assert 1, t.rows()==0
assert 2, t.cols()==4

@testing:case="test_MockData_function_futuresFactor_endDate_before_startDate", exception=1
futuresFactor(startDate=2021.12.31, endDate=2021.12.30, securityNumber=1, factorNumber=1)

@testing:case="test_MockData_function_futuresFactor_invalid_startDate_int", exception=1
futuresFactor(startDate=1, endDate=2021.12.31, securityNumber=1, factorNumber=1)

@testing:case="test_MockData_function_futuresFactor_invalid_securityNumber_string", exception=1
futuresFactor(startDate=2021.12.31, endDate=2021.12.31, securityNumber="1", factorNumber=1)

@testing:case="test_MockData_function_futuresFactor_invalid_factorNumber_string", exception=1
futuresFactor(startDate=2021.12.31, endDate=2021.12.31, securityNumber=1, factorNumber="1")
