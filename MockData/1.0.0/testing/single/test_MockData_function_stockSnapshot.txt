#include "setup/settings.txt"
use MockData

@testing:case="test_MockData_function_stockSnapshot_syntax"
re = syntax(stockSnapshot)
ex = "stockSnapshot(tradeDate,securityNumber)"
assert 1, eqObj(re, ex)

@testing:case="test_MockData_function_stockSnapshot_normal"
t = stockSnapshot(tradeDate=2020.01.06, securityNumber=10)
assert 1, form(t)==6
assert 2, t.rows()==48020
assert 3, t.cols()==62
assert 4, eqObj(t.schema().colDefs[0].name, "TradeDate")
assert 5, eqObj(t.schema().colDefs[0].typeInt, 6)
assert 6, eqObj(t.schema().colDefs[3].name, "Securityid")
assert 7, eqObj(t.schema().colDefs[3].typeInt, 17)

@testing:case="test_MockData_function_stockSnapshot_securityNumber_1"
t = stockSnapshot(tradeDate=2020.01.06, securityNumber=1)
assert 1, t.rows()==4802
assert 2, t.cols()==62

@testing:case="test_MockData_function_stockSnapshot_non_trading_day"
t = stockSnapshot(tradeDate=2020.01.05, securityNumber=10)
assert 1, t.rows()==0
assert 2, t.cols()==62

@testing:case="test_MockData_function_stockSnapshot_invalid_tradeDate_int", exception=1
stockSnapshot(tradeDate=1, securityNumber=10)

@testing:case="test_MockData_function_stockSnapshot_invalid_tradeDate_string", exception=1
stockSnapshot(tradeDate="2020.01.06", securityNumber=10)

@testing:case="test_MockData_function_stockSnapshot_invalid_securityNumber_string", exception=1
stockSnapshot(tradeDate=2020.01.06, securityNumber="10")

@testing:case="test_MockData_function_stockSnapshot_invalid_securityNumber_double", exception=1
stockSnapshot(tradeDate=2020.01.06, securityNumber=1.1)

@testing:case="test_MockData_function_stockSnapshot_invalid_securityNumber_NULL", exception=1
stockSnapshot(tradeDate=2020.01.06, securityNumber=NULL)

@testing:case="test_MockData_function_stockSnapshot_invalid_tradeDate_NULL", exception=1
stockSnapshot(tradeDate=NULL, securityNumber=10)
