#include "setup/settings.txt"
use MockData

@testing:case="test_MockData_function_dailyFactor_syntax"
re = syntax(dailyFactor)
ex = "dailyFactor(startDate,endDate,securityNumber,factorNumber)"
assert 1, eqObj(re, ex)

@testing:case="test_MockData_function_dailyFactor_normal"
t = dailyFactor(startDate=2021.01.04, endDate=2021.01.08, securityNumber=10, factorNumber=5)
assert 1, form(t)==6
assert 2, t.cols()==4
assert 3, eqObj(t.schema().colDefs[0].name, "tradetime")
assert 4, eqObj(t.schema().colDefs[0].typeInt, 6)
assert 5, eqObj(t.schema().colDefs[1].name, "securityid")
assert 6, eqObj(t.schema().colDefs[1].typeInt, 17)
assert 7, eqObj(t.schema().colDefs[2].name, "value")
assert 8, eqObj(t.schema().colDefs[2].typeInt, 16)
assert 9, eqObj(t.schema().colDefs[3].name, "factorname")
assert 10, eqObj(t.schema().colDefs[3].typeInt, 17)

@testing:case="test_MockData_function_dailyFactor_one_day_one_stock_one_factor"
tradeDates = getMarketCalendar("XSHG", 2021.01.04, 2021.01.04)
t = dailyFactor(startDate=2021.01.04, endDate=2021.01.04, securityNumber=1, factorNumber=1)
assert 1, t.rows()==1
assert 2, t.cols()==4

@testing:case="test_MockData_function_dailyFactor_non_trading_day_range"
t = dailyFactor(startDate=2021.01.01, endDate=2021.01.03, securityNumber=10, factorNumber=5)
assert 1, t.rows()==0
assert 2, t.cols()==4

@testing:case="test_MockData_function_dailyFactor_endDate_before_startDate", exception=1
dailyFactor(startDate=2021.01.08, endDate=2021.01.04, securityNumber=10, factorNumber=5)

@testing:case="test_MockData_function_dailyFactor_invalid_startDate_int", exception=1
dailyFactor(startDate=1, endDate=2021.01.08, securityNumber=10, factorNumber=5)

@testing:case="test_MockData_function_dailyFactor_invalid_securityNumber_string", exception=1
dailyFactor(startDate=2021.01.04, endDate=2021.01.08, securityNumber="10", factorNumber=5)

@testing:case="test_MockData_function_dailyFactor_invalid_factorNumber_string", exception=1
dailyFactor(startDate=2021.01.04, endDate=2021.01.08, securityNumber=10, factorNumber="5")

@testing:case="test_MockData_function_dailyFactor_invalid_factorNumber_double", exception=1
dailyFactor(startDate=2021.01.04, endDate=2021.01.08, securityNumber=10, factorNumber=1.1)
