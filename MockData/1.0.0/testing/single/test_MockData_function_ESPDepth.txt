#include "setup/settings.txt"
use MockData

@testing:case="test_MockData_function_ESPDepth_syntax"
re = syntax(ESPDepth)
ex = "ESPDepth(tradeDate,securityNumber)"
assert 1, eqObj(re, ex)

@testing:case="test_MockData_function_ESPDepth_normal"
t = ESPDepth(tradeDate=2020.06.01, securityNumber=10)
assert 1, form(t)==6
assert 2, t.rows()==800
assert 3, t.cols()==59
assert 4, eqObj(t.schema().colDefs[0].name, "tradeDate")
assert 5, eqObj(t.schema().colDefs[0].typeInt, 6)

@testing:case="test_MockData_function_ESPDepth_default_securityNumber"
t = ESPDepth(tradeDate=2020.06.01)
assert 1, form(t)==6
assert 2, t.cols()==59

@testing:case="test_MockData_function_ESPDepth_securityNumber_1"
t = ESPDepth(tradeDate=2020.06.01, securityNumber=1)
assert 1, t.rows()==80
assert 2, t.cols()==59

@testing:case="test_MockData_function_ESPDepth_non_trading_day"
t = ESPDepth(tradeDate=2020.06.07, securityNumber=10)
assert 1, t.rows()==0
assert 2, t.cols()==59

@testing:case="test_MockData_function_ESPDepth_invalid_tradeDate_int", exception=1
ESPDepth(tradeDate=1, securityNumber=10)

@testing:case="test_MockData_function_ESPDepth_invalid_tradeDate_string", exception=1
ESPDepth(tradeDate="2020.06.01", securityNumber=10)

@testing:case="test_MockData_function_ESPDepth_invalid_securityNumber_string", exception=1
ESPDepth(tradeDate=2020.06.01, securityNumber="10")

@testing:case="test_MockData_function_ESPDepth_invalid_securityNumber_double", exception=1
ESPDepth(tradeDate=2020.06.01, securityNumber=1.1)
