@testing:case="prepare_data_1"
existsTableCheck = getStreamTables().name
clearSTable = def (stName){
    try{
    	if (existsStreamTable(stName)){
        	dropStreamTable(stName)
    	}
    }catch(ex){}
}
each(clearSTable, existsTableCheck)
try{
	unsubscribeTable(,"st5","sub5")
	dropStreamTable("st5")
	}
catch(ex){print(ex)}

colName=["time","x"]
colType=["timestamp","int"]
t1 = streamTable(100:0, colName, colType);
share t1 as st1
go
sleep(1000)

@testing:case="test_function_getNonPersistenceTableStat_Not_Persistence"
re =  ops::getNonPersistenceTableStat()
assert 1, size(re) == 1
