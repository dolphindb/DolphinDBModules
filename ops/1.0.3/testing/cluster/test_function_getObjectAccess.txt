def checkFunctionView(uid, isAdmin, objectName, reTable){
    re = select * from reTable where userId=uid
    assert 1, isNull(re.canRead) && isNull(re.canInsert) && isNull(re.canUpdate) && isNull(re.canDelete)
    assert 2, re.canExec == true
    assert 3, re.userId == uid
    assert 4, re.objectType == "functionView"
    assert 5, re.objectName == objectName
    assert 6, re.isAdmin == isAdmin
    assert 7, size(re.userId)==1
}

def checkDatabase(uid, isAdmin, object, reTable, canRead, canInsert, canUpdate, canDelete){
    re = select * from reTable where userId=uid and objectName=object
    assert 1, isNull(re.canExec)
    assert 2, re.userId == uid
    assert 3, re.objectType == "database"
    assert 4, re.objectName == object
    assert 5, re.canRead == canRead
    assert 6, re.canInsert == canInsert
    assert 7, re.canUpdate == canUpdate
    assert 8, re.canDelete == canDelete
    assert 9, re.isAdmin == isAdmin
    assert 10, size(re.userId)==1
}

def checkTable(uid, isAdmin, object, reTable, canRead, canInsert, canUpdate, canDelete){
    re = select * from reTable where userId=uid and objectName=object
    assert 1, isNull(re.canExec)
    assert 2, re.userId == uid
    assert 3, re.objectType == "table"
    assert 4, re.objectName == object
    assert 5, re.canRead == canRead
    assert 6, re.canInsert == canInsert
    assert 7, re.canUpdate == canUpdate
    assert 8, re.canDelete == canDelete
    assert 9, re.isAdmin == isAdmin
    assert 10, size(re.userId)==1
}
