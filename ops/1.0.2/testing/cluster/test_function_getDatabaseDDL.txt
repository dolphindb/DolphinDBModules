use ops
def compare2Dict(dict1,dict2,excludeKeys=NULL){
	keyInDict1 = dict1.keys()[not dict1.keys() in array(STRING).append!(excludeKeys)]
	keyInDict2 = dict2.keys()[not dict2.keys() in array(STRING).append!(excludeKeys)]
	if(count(keyInDict1)!=count(keyInDict2)){
		return false
		}
	for(i in keyInDict1){
		//标量比较
		typeName = split((typestr(dict1[i]))," ")
		if(size(typeName)==1){
			if(dict1[i]!=dict2[i]){
				return false
				}
			}
		//向量比较
		else if(any(typeName==`VECTOR)){
			if(any(dict1[i]!=dict2[i])){
				return false
				}
			}
		//表比较
		else if(any(typeName==`TABLE)){
			if(!all(each(eqObj, dict1[i].values(), dict2[i].values()))){
				return false
				}
			}
		}
	return true
}
