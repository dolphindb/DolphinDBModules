def supportVersion(){
    ver = split(split(version()," ")[0],".")
    verL = int(ver[0])
    verM = int(ver[2])
    if(verL==2 and verM<11){
        return false
    }else{
        return true
    }
}
tradeDateVesion= supportVersion()
