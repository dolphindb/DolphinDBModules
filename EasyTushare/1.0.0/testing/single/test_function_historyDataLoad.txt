def runHistoryImport(token, historyFunc, sourceTable, expectedJobDesc, dbPath, targetTable, startDate, endDate){
    submittedAt = now()
    historyFunc(token, startDate, endDate, [sourceTable])
    jobs = exec jobId from getRecentJobs() where jobDesc == expectedJobDesc and receivedTime >= submittedAt
    if(jobs.size() != 1){
        return [long(-1), long(-1), long(-1), long(-1)]
    }
    getJobReturn(jobs[0], true)
    rowCount = exec count(*) from loadTable(dbPath, targetTable)
    currentImportCount = exec count(*) from loadTable(dbPath, targetTable) where update_time >= submittedAt
    nullUpdateTimeCount = exec count(*) from loadTable(dbPath, targetTable) where isNull(update_time)
    return [long(jobs.size()), long(rowCount), long(currentImportCount), long(nullUpdateTimeCount)]
}

ensureHistoryTestTables()
