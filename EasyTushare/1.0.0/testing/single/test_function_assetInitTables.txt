// Each case uses one representative table.  InitTables([]) is intentionally
// covered by the package integration job because it creates every table.
@testing:case="test_stockInitTables_selected"
if(!existsTable("dfs://tushare_basic_db", "stock_basic")){
    stockInitTables(["stock_basic"])
}
assert 1, existsTable("dfs://tushare_basic_db", "stock_basic")
assert 2, stockGetAllTableNames().size() > 0
assert 3, "stock_basic" in stockGetAllTableNames()
