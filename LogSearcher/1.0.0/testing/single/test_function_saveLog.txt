@testing:case = "prepare"
use LogSearcher
go

def LoadSaveLog(relog, eachpath){
	if(eachpath like "%job.log"){
		// job
		size_num = split(eachpath, "/").size()
		filename = split(eachpath, "/")[size_num-1]
		excontent = select content from relog where logType = "job" and like(logFile, "%/"+filename+"%")
		eachcontent = loadText(eachpath, delimiter = "\\n", containHeader = false)
	}else if(eachpath like "%hardware.log"){
		// hardware
		excontent = select content from relog where logType = "hardware"
		eachcontent = loadText(eachpath, delimiter = "\\n", containHeader = false)
	}else if(eachpath like "%aclAudit.log"){
		// aclAudit
		size_num = split(eachpath, "/").size()
		filename = split(eachpath, "/")[size_num-1]
		excontent = select content from relog where logType = "aclAudit" and like(logFile, "%/"+filename+"%")
		eachcontent = loadText(eachpath, delimiter = "\\n", containHeader = false)
	}else if(eachpath like "%query.log"){
		// query
		size_num = split(eachpath, "/").size()
		filename = split(eachpath, "/")[size_num-1]
		excontent = select content from relog where logType = "query" and like(logFile, "%/"+filename+"%")
		eachcontent = loadText(eachpath, delimiter = "\\n", containHeader = false)
	}else if(eachpath like "%audit.log"){
		// audit
		size_num = split(eachpath, "/").size()
		filename = split(eachpath, "/")[size_num-1]
		excontent = select content from relog where logType = "audit" and like(logFile, "%/"+filename+"%")
		eachcontent = loadText(eachpath, delimiter = "\\n", containHeader = false)
	}else if(eachpath like "%dolphindb.log"){
		// running
		size_num = split(eachpath, "/").size()
		filename = split(eachpath, "/")[size_num-1]
		excontent = select content from relog where logType = "running" and like(logFile, "%/"+filename+"%")
		eachcontent = loadText(eachpath, delimiter = "\\n", containHeader = false)
	}else if(eachpath like "%access.log"){
		// access
		size_num = split(eachpath, "/").size()
		filename = split(eachpath, "/")[size_num-1]
		filetime = split(filename, "_")
		if(filetime.size() > 2){
			excontent = select content from relog where logType = "access" and like(logFile, "%/"+filetime[0]+"%")
		}else{
			excontent = select content from relog where logType = "access" and like(logFile, "%/"+filetime[1]+"%")
		}
		eachcontent = loadText(eachpath, delimiter = "\\n", containHeader = false)
	}else if(eachpath like "%dmesg.log"){
		// access
		excontent = select content from relog where logType = "dmesg"
		eachcontent = loadText(eachpath, delimiter = "\\n", containHeader = false)
	}else if(eachpath like "%core%"){
		// coredump
		size_num = split(eachpath, "/").size()
		filename = split(eachpath, "/")[size_num-1]
		size_num2 = split(filename, ".").size()
		filename2 = concat(split(filename, ".")[0:(size_num2-2)], ".")
		excontent = select content from relog where logType = "coredump" and like(logFile, "%"+filename2+"%")
		fin = file(eachpath, "r")
		eachcontent = ""
		eachcontent = readLines(fin)
		fin.close()
		eachcontent = table(concat(eachcontent, "\n") as col0)
	}else{
		throw "no this type log"
	}
	return excontent, eachcontent
}

@testing:case = "test_function_saveLog_para_default"
tnow = now()
tstart = temporalAdd(tnow, -1, "d")
re = LogSearcher::searchLog({"dolphindbPath":dolphindbPath})
assert 1, re.size() > 0
logpath = LogSearcher::saveLog(re)
hostname = exec host from rpc(getControllerAlias(), getClusterPerf{true})
allpath = logpath + "/"+ hostname[0]+"/" +files(logpath + "/"+ hostname[0]).filename
for(eachpath in allpath){
	excontent, eachcontent = LoadSaveLog(re, eachpath)
	assert 2, each(eqObj, excontent.values(), eachcontent.values())
}
// clear
try{
	rmdir(logpath, true)
}catch(exp){print(exp)}
