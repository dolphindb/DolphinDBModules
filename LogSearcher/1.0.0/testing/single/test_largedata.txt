def LoadSaveLog(relog, eachpath){
	if(eachpath like "%job.log"){
		// job
		excontent = select content from relog where logType = "job"
		eachcontent = loadText(eachpath, delimiter = "\\n", containHeader = false)
	}else if(eachpath like "%hardware.log"){
		// hardware
		excontent = select content from relog where logType = "hardware"
		eachcontent = loadText(eachpath, delimiter = "\\n", containHeader = false)
	}else if(eachpath like "%aclAudit.log"){
		// aclAudit
		excontent = select content from relog where logType = "aclAudit"
		eachcontent = loadText(eachpath, delimiter = "\\n", containHeader = false)
	}else if(eachpath like "%query.log"){
		// query
		excontent = select content from relog where logType = "query"
		eachcontent = loadText(eachpath, delimiter = "\\n", containHeader = false)
	}else if(eachpath like "%audit.log"){
		// audit
		excontent = select content from relog where logType = "audit"
		eachcontent = loadText(eachpath, delimiter = "\\n", containHeader = false)
	}else if(eachpath like "%dolphindb.log"){
		// running
		excontent = select content from relog where logType = "running"
		eachcontent = loadText(eachpath, delimiter = "\\n", containHeader = false)
	}else if(eachpath like "%access.log"){
		// access
		excontent = select content from relog where logType = "access"
		eachcontent = loadText(eachpath, delimiter = "\\n", containHeader = false)
	}else if(eachpath like "%core%"){
		// coredump
		excontent = select content from re where logType = "coredump"
		fin = file(eachpath, "r")
		eachcontent = ""
		eachcontent = readLines(fin)
		fin.close()
		eachcontent = table(concat(eachcontent, "\n") as col0)
	}else{
		throw "no this type log"
	}
	return excontent, eachcontent
}
