tstart = 2025.06.09 00:00:00.000
tend = temporalAdd(tnow, 1, "d")
para = {"cacheSize":64, "startTime":tstart, "endTime":tend, "logType":["running", "job", "hardware", "query", "audit", "aclAudit", "access", "hardware"]}
para["dolphindbPath"] = dolphindbPath
re = LogSearcher::searchLog(para)
re_cache = select * from getRecentJobs() where receivedTime >= tnow
assert 1, re.size() > 0
assert 2, re_cache.size() > 0
assert 3, re.timestamp >= tstart
assert 4, re.timestamp <= tend
re2 = getJobReturn(re_cache.jobId[0], true)
assert 5, each(eqObj, re.values(), re2.values())
re3 = exec bytes\1024\1024 from objs() where name = "re"
assert 6, re3 < 64

@testing:case = "test_function_searchLog_multipara"
tnow = now()
tstart = 2025.06.09 00:00:00.000
tend = temporalAdd(tnow, 1, "d")
para = dict(STRING, ANY)
para["pattern"] = ["%ERROR%", "%WARNING%", "%dfs:%"]
para["startTime"] = tstart
para["endTime"] = tend
para["logType"] = ["running", "query", "audit"]
para["node"] = getNodeAlias()
para["mode"] = "ilike"
para["cacheSize"] = -1
para["adminPassword"] = "123456"
para["dolphindbPath"] = dolphindbPath
re = LogSearcher::searchLog(para)
re_cache = select * from getRecentJobs() where receivedTime >= tnow
assert 1, re.size() > 0
assert 2, re_cache.size() > 0
assert 3, re.timestamp >= tstart
assert 4, re.timestamp <= tend
re2 = getJobReturn(re_cache.jobId[0], true)
assert 5, each(eqObj, re.values(), re2.values())
assert 6, exec ilike(content, "%ERROR%") or ilike(content, "%WARNING%") or ilike(content, "%dfs:%") as a1 from re
assert 7, distinct(re.logType).sort() == ["running", "query", "audit"].sort()
assert 8, distinct(re.logLevel).sort() == ["INFO", "ERROR", "WARNING", "DEBUG"].sort()
assert 9, distinct(re.node).sort() == [getNodeAlias()].sort()
re3 = select * from re where logType = "query"
assert 10, strFind(re3.logFile, "query") != -1
re4 = select * from re where logType = "audit"
assert 11, strFind(re4.logFile, "audit") != -1
